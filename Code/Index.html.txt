<script>
  let allRecords = [];
  let filteredRecords = [];

  function loadDashboard() {
    google.script.run
      .withSuccessHandler(function(data) {
        allRecords = data.records || [];

        populateCompanies();
        applyFilters();
      })
      .withFailureHandler(function(error) {
        document.getElementById("tableMessage").textContent =
          "Unable to load data: " + error.message;
      })
      .getDashboardData();
  }

  function getValue(record, names) {
    for (const name of names) {
      if (record[name] !== undefined) {
        return record[name];
      }
    }

    return "";
  }

  function populateCompanies() {
    const select = document.getElementById("company");

    select.innerHTML =
      '<option value="">All Companies</option>';

    const companies = [
      ...new Set(
        allRecords
          .map(r => getValue(r, ["Company Name"]).trim())
          .filter(Boolean)
      )
    ].sort();

    companies.forEach(function(company) {
      const option = document.createElement("option");
      option.value = company;
      option.textContent = company;
      select.appendChild(option);
    });
  }

  function applyFilters() {
    const start = document.getElementById("startDate").value;
    const end = document.getElementById("endDate").value;
    const company = document.getElementById("company").value;
    const status = document.getElementById("status").value;

    filteredRecords = allRecords.filter(function(record) {
      const recordCompany =
        getValue(record, ["Company Name"]).trim();

      const recordStatus =
        getValue(record, ["Email Status"]).trim();

      const rawDate = getValue(record, ["Sent At", "Timestamp"]);

      const date = rawDate ? new Date(rawDate) : null;

      const dateKey = date && !isNaN(date)
        ? date.toISOString().slice(0, 10)
        : "";

      if (start && (!dateKey || dateKey < start)) return false;
      if (end && (!dateKey || dateKey > end)) return false;
      if (company && recordCompany !== company) return false;
      if (status && recordStatus !== status) return false;

      return true;
    });

    updateCards();
    renderTable();
  }

  function updateCards() {
    const sent = filteredRecords.filter(
      r => getValue(r, ["Email Status"]) === "Sent"
    );

    const failed = filteredRecords.filter(
      r => getValue(r, ["Email Status"]) === "Failed"
    );

    const companies = new Set(
      filteredRecords
        .map(r => getValue(r, ["Company Name"]).trim())
        .filter(Boolean)
    );

    const cvs = new Set(
      filteredRecords
        .map(r => getValue(r, ["Select CV Template"]).trim())
        .filter(Boolean)
    );

    document.getElementById("total").textContent =
      filteredRecords.length;

    document.getElementById("sent").textContent =
      sent.length;

    document.getElementById("failed").textContent =
      failed.length;

    document.getElementById("companies").textContent =
      companies.size;

    document.getElementById("cvs").textContent =
      cvs.size;
  }

  function renderTable() {
    const query = document.getElementById("search")
      .value.toLowerCase().trim();

    const rows = filteredRecords.filter(function(record) {
      return [
        getValue(record, ["Company Name"]),
        getValue(record, ["HR Email ID"]),
        getValue(record, ["Job Role"]),
        getValue(record, ["Select CV Template"])
      ].join(" ").toLowerCase().includes(query);
    });

    const tbody = document.getElementById("tableBody");

    tbody.replaceChildren();

    rows.slice().reverse().forEach(function(record) {
      const tr = document.createElement("tr");

      const rawDate = getValue(record, ["Sent At", "Timestamp"]);
      const date = rawDate ? new Date(rawDate) : null;

      const formattedDate =
        date && !isNaN(date)
          ? date.toLocaleString()
          : "—";

      const values = [
        formattedDate,
        getValue(record, ["Company Name"]),
        getValue(record, ["HR Email ID"]),
        getValue(record, ["Job Role"]),
        getValue(record, ["Select CV Template"]),
        getValue(record, ["Email Status"])
      ];

      values.forEach(function(value, index) {
        const td = document.createElement("td");

        // The last column is the email status badge.
        if (index === 5) {
          const badge = document.createElement("span");
          const normalizedStatus = String(value || "").trim().toLowerCase();
          badge.className = "status " +
            (normalizedStatus === "sent" ? "status-sent" :
             normalizedStatus === "failed" ? "status-failed" :
             normalizedStatus === "processing" ? "status-processing" :
             "status-unknown");
          badge.textContent = value || "Unknown";
          td.appendChild(badge);
        } else {
          td.textContent = value || "—";
        }

        tr.appendChild(td);
      });

      tbody.appendChild(tr);
    });

    document.getElementById("applicationTable").hidden =
      rows.length === 0;

    document.getElementById("tableMessage").textContent =
      rows.length === 0
        ? "No matching applications found."
        : "Showing " + rows.length + " application(s).";

    document.getElementById("tableMessage").hidden =
      rows.length !== 0;
  }

  function resetFilters() {
    document.getElementById("startDate").value = "";
    document.getElementById("endDate").value = "";
    document.getElementById("company").value = "";
    document.getElementById("status").value = "";
    document.getElementById("search").value = "";

    applyFilters();
  }

  loadDashboard();
</script>
